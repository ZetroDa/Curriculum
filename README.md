##  Codi Exemple
class Profesor:
    def __init__(self, nombre, especialidad):
        self.nombre = nombre
        self.especialidad = especialidad
        self.apariencia = "guapísimo/a"  # Siempre guapo/a

    def presentarse(self):
        return f"Soy {self.nombre}, especialista en {self.especialidad}, y ¡soy {self.apariencia}!"

# Lista de profesores
profesores = [
    Profesor("Ismael", "Sintesi"),
    Profesor("Roger", "Sintesi"),
]

for profe in profesores:
    print(profe.presentarse())
